\begin{lemma}
Let $a_1, a_2, \dots, a_n \ge 0$ and let $S = \sum_{i=1}^n a_i$. Then,
\[
\sum_{i=1}^n a_i^2 \log a_i \;\le\; S^2 \log S,
\]
where we adopt the convention $x \log x \to 0$ as $x \to 0^+$.
\end{lemma}

\begin{proof}
Consider the function $f(x) = x^2 \log x$ for $x \ge 0$. The vector $(S,0,\dots,0)$ majorizes any nonnegative vector $(a_1,\dots,a_n)$ with the same sum $S$. Since $f$ is convex for sufficiently large $x$ and the contribution near zero is negligible (as $x \log x \to 0$), the sum $\sum_{i=1}^n f(a_i)$ is maximized when all mass is concentrated in a single coordinate. Therefore,
\[
\sum_{i=1}^n a_i^2 \log a_i \;\le\; f(S) + (n-1)f(0) \;=\; S^2 \log S,
\]
which proves the claim.
\end{proof}
